#include <cstdint>
#include <cstring>

enum PixelFlag : uint16_t { SpectraMask=31, HasUp=32, HasDown=64, HasLeft=128, HasRight=256 };

static inline void enqueue(uint32_t index, uint16_t active, uint32_t* queue, uint16_t* flags,
                           uint32_t& tail, uint32_t& count, uint32_t capacity) {
    if (!active) return;
    if (!(flags[index] & SpectraMask)) {
        queue[tail] = index;
        if (++tail == capacity) tail = 0;
        ++count;
    }
    flags[index] |= active;
}

template<uint32_t Levels>
static inline uint16_t update(float* out, const float* mask, uint32_t neighbor,
                          const float (&source)[Levels], uint16_t active) {
    float* target = out + static_cast<size_t>(neighbor) * Levels;
    const float bound = mask[neighbor];
    uint16_t changed = 0;
    for (uint32_t k=0; k<Levels; ++k) {
        if (!(active & (1u << k))) continue;
        const float value = source[k] > bound ? bound : source[k];
        const bool raised = !(value <= target[k]);
        target[k] = raised ? value : target[k];
        changed |= raised ? (1u << k) : 0;
    }
    return changed;
}

template<uint32_t Levels>
static void reconstruct(const float* __restrict seed, const float* __restrict mask,
                        float* __restrict out, uint32_t height, uint32_t width,
                        uint32_t* __restrict queue, uint16_t* __restrict flags, uint64_t* counters) {
    const uint32_t capacity = height * width;
    std::memcpy(out, seed, static_cast<size_t>(capacity) * Levels * sizeof(float));
    for (uint32_t row=0; row<height; ++row) {
        const uint32_t base = row * width;
        const uint16_t row_flags = (row>0 ? HasUp : 0) | (row+1<height ? HasDown : 0);
        for (uint32_t col=0; col<width; ++col) {
            const uint32_t index = base + col;
            flags[index] = row_flags | (col>0 ? HasLeft : 0) | (col+1<width ? HasRight : 0);
            const float bound = mask[index];
            float* values = out + static_cast<size_t>(index) * Levels;
            for (uint32_t k=0; k<Levels; ++k) {
                float value = values[k];
                if (row>0 && values[static_cast<int64_t>(k)-static_cast<int64_t>(width)*Levels]>value) value = values[static_cast<int64_t>(k)-static_cast<int64_t>(width)*Levels];
                if (col>0 && values[static_cast<int64_t>(k)-Levels]>value) value = values[static_cast<int64_t>(k)-Levels];
                if (value>bound) value = bound;
                values[k] = value;
            }
        }
    }
    uint32_t head=0, tail=0, count=0;
    for (uint32_t row=height; row-- > 0;) {
        const uint32_t base = row * width;
        for (uint32_t col=width; col-- > 0;) {
            const uint32_t index = base + col;
            const float bound = mask[index];
            float* values = out + static_cast<size_t>(index) * Levels;
            uint16_t active = 0;
            for (uint32_t k=0; k<Levels; ++k) {
                float value = values[k];
                if (row+1<height && values[k+width*Levels]>value) value = values[k+width*Levels];
                if (col+1<width && values[k+Levels]>value) value = values[k+Levels];
                if (value>bound) value = bound;
                values[k] = value;
                if (row>0 && values[static_cast<int64_t>(k)-static_cast<int64_t>(width)*Levels]<value && values[static_cast<int64_t>(k)-static_cast<int64_t>(width)*Levels]<mask[index-width]) active |= (1u << k);
                if (row+1<height && values[k+width*Levels]<value && values[k+width*Levels]<mask[index+width]) active |= (1u << k);
                if (col>0 && values[static_cast<int64_t>(k)-Levels]<value && values[static_cast<int64_t>(k)-Levels]<mask[index-1]) active |= (1u << k);
                if (col+1<width && values[k+Levels]<value && values[k+Levels]<mask[index+1]) active |= (1u << k);
            }
            enqueue(index,active,queue,flags,tail,count,capacity);
        }
    }
    while (count>0) {
        const uint32_t index = queue[head];
        const uint16_t state = flags[index];
        const uint16_t active = state & SpectraMask;
        ++counters[0];
        counters[1] += __builtin_popcount(active);
        flags[index] = state & ~SpectraMask;
        if (++head==capacity) head=0;
        --count;
        float source[Levels];
        std::memcpy(source,out+static_cast<size_t>(index)*Levels,sizeof(source));
        if (state&HasUp) enqueue(index-width,update(out,mask,index-width,source,active),queue,flags,tail,count,capacity);
        if (state&HasDown) enqueue(index+width,update(out,mask,index+width,source,active),queue,flags,tail,count,capacity);
        if (state&HasLeft) enqueue(index-1,update(out,mask,index-1,source,active),queue,flags,tail,count,capacity);
        if (state&HasRight) enqueue(index+1,update(out,mask,index+1,source,active),queue,flags,tail,count,capacity);
    }
}

extern "C" void reconstruct_five(const float* seed, const float* mask, float* out,
                                 uint32_t height, uint32_t width,
                                 uint32_t* queue, uint16_t* flags, uint64_t* counters) {
    counters[0]=0; counters[1]=0;
    reconstruct<5>(seed,mask,out,height,width,queue,flags,counters);
}
