from pathlib import Path
from nominal_refactor_advisor.codemod import (
    CodemodPlanDocument, CodemodSourceSnapshot, PatchTargetOperation,
    RefactorRecipe, SourceRewriteTarget, SourceTextReplacement,
)

root=Path('/home/ts/code/projects/openhcs-compile-perf')
sources={}; changes={}
path=root/'openhcs/core/process_local_cache.py'
original=path.read_text(); sources[str(path)]=original
text=original.replace('class ProcessLocalBoundedCache(Generic[CacheKey, CachedValue]):\n    """Bounded LRU cache with one singleton instance per concrete subclass."""',
    'class BoundedCache(Generic[CacheKey, CachedValue]):\n    """Bounded LRU values with lifetime supplied by their owning consumer."""',1)
anchor='    @classmethod\n    def process_cache(cls) -> "ProcessLocalBoundedCache[CacheKey, CachedValue]":'
assert text.count(anchor)==1
text=text.replace(anchor, '''
@dataclass(slots=True)
class ProcessLocalBoundedCache(BoundedCache[CacheKey, CachedValue]):
    """Bounded values with one process-local singleton per concrete subclass."""

    @classmethod
    def process_cache(cls) -> "ProcessLocalBoundedCache[CacheKey, CachedValue]":''',1)
changes[str(path)]=text
for relative in ('openhcs/core/runtime_stores.py','openhcs/core/measurement_feature_queries.py','openhcs/core/runtime_artifact_queries.py'):
    path=root/relative; original=path.read_text(); sources[str(path)]=original
    text=original.replace('ProcessLocalBoundedCache','BoundedCache')
    # Existing registered process-cache families retain their process lifecycle.
    text=text.replace('RegisteredBoundedCache','RegisteredProcessLocalBoundedCache')
    changes[str(path)]=text
operations=tuple(PatchTargetOperation(target=SourceRewriteTarget(file_path=path),replacements=(SourceTextReplacement(old_source=sources[path],new_source=text),),rationale='Existing cache storage operations move intact to the common value owner; process singleton methods stay on their process owner.') for path,text in changes.items())
plan=CodemodPlanDocument(recipes=(RefactorRecipe(recipe_id='separate-cache-algorithm-and-lifetime',reason='Store-bound cache domains must inherit values without also inheriting a competing process-singleton API.',operations=operations),))
simulation=plan.simulate(CodemodSourceSnapshot.from_source_mapping(sources))
assert simulation.is_clean,simulation.simulation_payload()
Path('/home/ts/code/projects/openhcs-benchmark-runs/perf-generic-runtime-cache-lifetime-refinement-20260930.diff').write_text(simulation.unified_diff(sources))
print(simulation.apply())
