#include <cstdint>
#include <cstring>

enum PixelFlag : uint8_t { Queued=1, HasUp=2, HasDown=4, HasLeft=8, HasRight=16 };

static inline void enqueue(uint32_t index, uint32_t* queue, uint8_t* flags,
                           uint32_t& tail, uint32_t& count, uint32_t capacity) {
    if (flags[index] & Queued) return;
    flags[index] |= Queued;
    queue[tail] = index;
    if (++tail == capacity) tail = 0;
    ++count;
}

template<uint32_t Levels>
static inline bool update(float* out, const float* mask, uint32_t neighbor,
                          const float (&source)[Levels]) {
    float* target = out + static_cast<size_t>(neighbor) * Levels;
    const float bound = mask[neighbor];
    bool changed = false;
    for (uint32_t k=0; k<Levels; ++k) {
        const float value = source[k] > bound ? bound : source[k];
        const bool raised = !(value <= target[k]);
        target[k] = raised ? value : target[k];
        changed |= raised;
    }
    return changed;
}

template<uint32_t Levels>
static void reconstruct(const float* __restrict seed, const float* __restrict mask,
                        float* __restrict out, uint32_t height, uint32_t width,
                        uint32_t* __restrict queue, uint8_t* __restrict flags) {
    const uint32_t capacity = height * width;
    std::memcpy(out, seed, static_cast<size_t>(capacity) * Levels * sizeof(float));
    for (uint32_t row=0; row<height; ++row) {
        const uint32_t base = row * width;
        const uint8_t row_flags = (row>0 ? HasUp : 0) | (row+1<height ? HasDown : 0);
        for (uint32_t col=0; col<width; ++col) {
            const uint32_t index = base + col;
            flags[index] = row_flags | (col>0 ? HasLeft : 0) | (col+1<width ? HasRight : 0);
            const float bound = mask[index];
            float* values = out + static_cast<size_t>(index) * Levels;
            for (uint32_t k=0; k<Levels; ++k) {
                float value = values[k];
                if (row>0 && values[static_cast<int64_t>(k)-static_cast<int64_t>(width)*Levels]>value) value = values[static_cast<int64_t>(k)-static_cast<int64_t>(width)*Levels];
                if (col>0 && values[static_cast<int64_t>(k)-Levels]>value) value = values[static_cast<int64_t>(k)-Levels];
                if (value>bound) value = bound;
                values[k] = value;
            }
        }
    }
    uint32_t head=0, tail=0, count=0;
    for (uint32_t row=height; row-- > 0;) {
        const uint32_t base = row * width;
        for (uint32_t col=width; col-- > 0;) {
            const uint32_t index = base + col;
            const float bound = mask[index];
            float* values = out + static_cast<size_t>(index) * Levels;
            bool can_raise = false;
            for (uint32_t k=0; k<Levels; ++k) {
                float value = values[k];
                if (row+1<height && values[k+width*Levels]>value) value = values[k+width*Levels];
                if (col+1<width && values[k+Levels]>value) value = values[k+Levels];
                if (value>bound) value = bound;
                values[k] = value;
                if (row>0 && values[static_cast<int64_t>(k)-static_cast<int64_t>(width)*Levels]<value && values[static_cast<int64_t>(k)-static_cast<int64_t>(width)*Levels]<mask[index-width]) can_raise=true;
                if (row+1<height && values[k+width*Levels]<value && values[k+width*Levels]<mask[index+width]) can_raise=true;
                if (col>0 && values[static_cast<int64_t>(k)-Levels]<value && values[static_cast<int64_t>(k)-Levels]<mask[index-1]) can_raise=true;
                if (col+1<width && values[k+Levels]<value && values[k+Levels]<mask[index+1]) can_raise=true;
            }
            if (can_raise) enqueue(index,queue,flags,tail,count,capacity);
        }
    }
    while (count>0) {
        const uint32_t index = queue[head];
        const uint8_t state = flags[index];
        flags[index] = state & ~Queued;
        if (++head==capacity) head=0;
        --count;
        float source[Levels];
        std::memcpy(source,out+static_cast<size_t>(index)*Levels,sizeof(source));
        if ((state&HasUp) && update(out,mask,index-width,source)) enqueue(index-width,queue,flags,tail,count,capacity);
        if ((state&HasDown) && update(out,mask,index+width,source)) enqueue(index+width,queue,flags,tail,count,capacity);
        if ((state&HasLeft) && update(out,mask,index-1,source)) enqueue(index-1,queue,flags,tail,count,capacity);
        if ((state&HasRight) && update(out,mask,index+1,source)) enqueue(index+1,queue,flags,tail,count,capacity);
    }
}

extern "C" void reconstruct_five(const float* seed, const float* mask, float* out,
                                 uint32_t height, uint32_t width,
                                 uint32_t* queue, uint8_t* flags) {
    reconstruct<5>(seed,mask,out,height,width,queue,flags);
}
