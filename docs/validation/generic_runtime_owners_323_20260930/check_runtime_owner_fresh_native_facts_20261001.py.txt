import json
from pathlib import Path

from benchmark.adapters.openhcs import _strict_cellprofiler_runtime_equivalence_policy
from openhcs.core.equivalence.outputs import RuntimeOutputSnapshot, RuntimeTableNamespaceAdapter
from openhcs.core.equivalence.tables import RuntimeTableSnapshot
from openhcs.core.runtime_equivalence import RuntimeMeasurementSnapshot, runtime_measurement_equivalence

runs = Path('/home/ts/code/projects/openhcs-benchmark-runs')
native = runs / 'perf-runtime-owner-native-imaging-1w1j-20261001/native_jobs/0'
candidate_root = runs / 'perf-numerical-cache-owners-public-ExampleImagingFlowCytometryObjectsInGrid-1-candidate-20261001'
(candidate_path,) = tuple(candidate_root.rglob('BF_cells_on_grid.csv'))
policy = _strict_cellprofiler_runtime_equivalence_policy()
def snapshot(path):
    output = RuntimeOutputSnapshot(tables=RuntimeTableNamespaceAdapter.normalize((RuntimeTableSnapshot.from_csv(path),)))
    return RuntimeMeasurementSnapshot.from_output_snapshot(output, policy=policy)
candidate = snapshot(candidate_path)
checks = []
for repetition in (-1, 0, 1):
    reference = snapshot(native / str(repetition) / 'BF_cells_on_grid.csv')
    report = runtime_measurement_equivalence(reference, candidate, policy=policy)
    checks.append(dict(repetition=repetition,
                       reference_features=len(reference.measurement_fact_counts),
                       candidate_features=len(candidate.measurement_fact_counts),
                       differences=report.failure_messages()))
    print(json.dumps(dict(repetition=repetition, difference_count=len(report.differences), sample=report.failure_messages()[:12])), flush=True)
receipt = dict(source_revision='95a98b0083aba2384c271e224746824fc9e40ce5',
               scope='Existing strict measurement-fact comparison of the exported BF_cells_on_grid table; no image or ROI claim.',
               numeric_abs_tolerance=1e-6, numeric_rel_tolerance=1e-6, checks=checks)
(runs / 'perf-runtime-owner-fresh-native-fact-parity-20261001.json').write_text(json.dumps(receipt, indent=2) + '\n')
assert not any(check['differences'] for check in checks)
