import ast
import json
import re
import subprocess
from pathlib import Path

from nominal_refactor_advisor.ast_tools import SourceModule, module_syntax_index
from nominal_refactor_advisor.codemod import (
    CodemodPlanDocument, CodemodSourceSnapshot, PatchTargetOperation,
    RefactorRecipe, SourceRewriteTarget, SourceTextReplacement,
)

root = Path('/home/ts/code/projects/openhcs-compile-perf')
runs = root.parent / 'openhcs-benchmark-runs'
sources = {}; changes = {}
for relative in subprocess.check_output(['git','diff','--name-only','--diff-filter=AM'], cwd=root, text=True).splitlines():
    if not relative.endswith('.py'): continue
    path = root / relative
    original = path.read_text()
    text = original
    syntax = module_syntax_index(SourceModule.from_source_path(path, text).parse().module)
    imports = [node for _,node in syntax.indexed_nodes_of_type(ast.ImportFrom)
               if node.module == 'openhcs.core.measurement_row_materialization']
    if relative.startswith('openhcs/processing/backends/cellprofiler/') and len(imports) == 2:
        moved, = [node for node in imports if [alias.name for alias in node.names] == ['ObjectMeasurementColumnarRows']]
        existing, = [node for node in imports if node is not moved]
        lines = text.splitlines(keepends=True)
        removed = ''.join(lines[moved.lineno-1:moved.end_lineno])
        existing_text = ''.join(lines[existing.lineno-1:existing.end_lineno])
        assert existing_text.endswith(')\n')
        replacement = existing_text[:-2] + '    ObjectMeasurementColumnarRows,\n)\n'
        text = text.replace(existing_text, replacement, 1).replace(removed, '', 1)
    if relative == 'openhcs/core/measurement_row_materialization.py':
        text = text.replace('        return self.row_count()\n\n\n    def __iter__', '        return self.row_count()\n\n    def __iter__', 1)
        text = text.replace('        yield from self.iter_row_mappings()\n\n\n\nclass MeasurementRowDeclaredValue', '        yield from self.iter_row_mappings()\n\n\nclass MeasurementRowDeclaredValue', 1)
    # Black only the added/changed source ranges; keep unrelated legacy formatting.
    diff = subprocess.check_output(['git','diff','--unified=0','--',relative], cwd=root, text=True)
    ranges = []
    for match in re.finditer(r'^@@ -\d+(?:,\d+)? \+(\d+)(?:,(\d+))? @@', diff, re.M):
        start = int(match[1]); count = int(match[2] or 1)
        if count: ranges.append((max(1,start-3), start+count+3))
    if ranges:
        command = ['/home/ts/code/projects/openhcs/.venv/bin/python', '-m', 'black', '--quiet', '--target-version', 'py311']
        for start, end in ranges:
            command.extend(['--line-ranges', f'{start}-{end}'])
        command.extend(['--stdin-filename', str(path), '-'])
        text = subprocess.check_output(command, input=text, text=True, cwd=root)
    if text != original:
        sources[str(path)] = original; changes[str(path)] = text
operations = tuple(PatchTargetOperation(target=SourceRewriteTarget(file_path=path), replacements=(SourceTextReplacement(old_source=sources[path], new_source=text),), rationale='Consolidate the moved owner import and format only modified ranges; preserve semantic assertions.') for path,text in changes.items())
plan = CodemodPlanDocument(recipes=(RefactorRecipe(recipe_id='close-generic-measurement-import-migration', reason='All consumers use one nominal core import, without duplicated forwarding declarations.', operations=operations),))
simulation = plan.simulate(CodemodSourceSnapshot.from_source_mapping(sources))
assert simulation.is_clean, simulation.simulation_payload()
(runs / 'perf-generic-measurement-runtime-refinement-20260930.diff').write_text(simulation.unified_diff(sources))
print(simulation.apply())
