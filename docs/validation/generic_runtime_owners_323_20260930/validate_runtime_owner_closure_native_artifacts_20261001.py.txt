import hashlib
import json
import zipfile
from pathlib import Path

import numpy as np
import tifffile
from polystore.roi import load_rois_from_zip

runs = Path('/home/ts/code/projects/openhcs-benchmark-runs')
observations = json.loads((runs / 'perf-numerical-cache-owners-public-observations-20261001.json').read_text())
report = []


def normalized_run_addresses(value, run_root):
    """Change only the explicitly different, admitted run-directory prefix."""
    prefix = str(run_root) + '/'
    if isinstance(value, str):
        return '<run>/' + value[len(prefix):] if value.startswith(prefix) else value
    if isinstance(value, list):
        return [normalized_run_addresses(item, run_root) for item in value]
    if isinstance(value, dict):
        return {key: normalized_run_addresses(item, run_root) for key, item in value.items()}
    return value


for case in ('ExampleImagingFlowCytometryObjectsInGrid', 'cp_tutorial_3d_monolayer'):
    selected = [row for row in observations if row['case'] == case]
    assert [row['label'] for row in selected] == ['main', 'candidate']
    directories = [Path(row['directory']) for row in selected]
    image_root = ('images_' + case + '_source_workspace_well_throughput'
                  if case != 'cp_tutorial_3d_monolayer'
                  else 'images_3d_monolayer_final_source_workspace_well_throughput')
    folders = [directory / case / 'wells_1/workers_1' / image_root for directory in directories]
    inventories = [{path.relative_to(folder): path for path in folder.rglob('*')
                    if path.is_file() and path.suffix.lower() in ('.tif', '.tiff')}
                   for folder in folders]
    assert inventories[0].keys() == inventories[1].keys()
    images = []
    for name, control in sorted(inventories[0].items()):
        candidate = inventories[1][name]
        with tifffile.TiffFile(control) as base, tifffile.TiffFile(candidate) as head:
            assert len(base.series) == len(head.series), name
            for b, h in zip(base.series, head.series, strict=True):
                assert b.shape == h.shape and b.dtype == h.dtype and b.axes == h.axes, name
                assert np.array_equal(b.asarray(), h.asarray(), equal_nan=True), name
            assert len(base.pages) == len(head.pages), name
            for b, h in zip(base.pages, head.pages, strict=True):
                for tag in ('XResolution', 'YResolution', 'ResolutionUnit'):
                    bv = b.tags[tag].value if tag in b.tags else None
                    hv = h.tags[tag].value if tag in h.tags else None
                    assert bv == hv, (name, tag)
                assert bool(b.description) == bool(h.description), name
                if b.description:
                    bm = normalized_run_addresses(json.loads(b.description), directories[0])
                    hm = normalized_run_addresses(json.loads(h.description), directories[1])
                    assert bm == hm, (name, 'TIFF description/source metadata')
        images.append(str(name))
    roi_inventories = [{path.relative_to(folder): path for path in folder.rglob('*.roi.zip')}
                      for folder in folders]
    assert roi_inventories[0].keys() == roi_inventories[1].keys()
    archives = []
    for name, control in sorted(roi_inventories[0].items()):
        candidate = roi_inventories[1][name]
        base_rois, head_rois = load_rois_from_zip(control), load_rois_from_zip(candidate)
        assert len(base_rois) == len(head_rois), name
        with zipfile.ZipFile(control) as base, zipfile.ZipFile(candidate) as head:
            assert set(base.namelist()) == set(head.namelist()), name
            members = []
            for member in sorted(base.namelist()):
                b, h = base.read(member), head.read(member)
                if member.endswith('.json'):
                    assert normalized_run_addresses(json.loads(b), directories[0]) == normalized_run_addresses(json.loads(h), directories[1]), (name, member)
                else:
                    assert b == h, (name, member)
                members.append(member)
        archives.append({'path': str(name), 'native_roi_count': len(base_rois), 'members': len(members)})
    report.append({'case': case, 'main_revision': selected[0]['source_revision'],
                   'candidate_revision': selected[1]['source_revision'],
                   'exact_images': images, 'exact_native_roi_archives': archives,
                   'normalization': 'Only explicit run-directory prefixes in JSON source addresses; ZIP container timestamps are not scientific content. All native ROI members are byte-exact; all TIFF series pixels/dtype/shape/axes and physical/source metadata match.'})

destination = runs / 'perf-runtime-owner-closure-native-artifact-parity-20261001.json'
destination.write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps({'exact_images': sum(len(row['exact_images']) for row in report),
                  'exact_native_roi_archives': sum(len(row['exact_native_roi_archives']) for row in report),
                  'native_rois': sum(archive['native_roi_count'] for row in report for archive in row['exact_native_roi_archives'])}))
