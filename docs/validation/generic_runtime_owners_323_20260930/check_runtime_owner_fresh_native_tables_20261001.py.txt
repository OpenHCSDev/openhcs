import hashlib
import json
from pathlib import Path

from benchmark.adapters.openhcs import _strict_cellprofiler_runtime_equivalence_policy
from openhcs.core.equivalence.comparison import runtime_table_differences
from openhcs.core.equivalence.outputs import RuntimeTableNamespaceAdapter
from openhcs.core.equivalence.tables import RuntimeTableSnapshot

runs = Path('/home/ts/code/projects/openhcs-benchmark-runs')
native = runs / 'perf-runtime-owner-native-imaging-1w1j-20261001/native_jobs/0'
candidate_root = runs / 'perf-numerical-cache-owners-public-ExampleImagingFlowCytometryObjectsInGrid-1-candidate-20261001'
(candidate_path,) = tuple(candidate_root.rglob('BF_cells_on_grid.csv'))
candidate = RuntimeTableNamespaceAdapter.normalize((RuntimeTableSnapshot.from_csv(candidate_path),))
policy = _strict_cellprofiler_runtime_equivalence_policy()
checks = []
for repetition in (-1, 0, 1):
    path = native / str(repetition) / 'BF_cells_on_grid.csv'
    reference = RuntimeTableNamespaceAdapter.normalize((RuntimeTableSnapshot.from_csv(path),))
    differences = runtime_table_differences(reference, candidate, policy)
    checks.append(dict(repetition=repetition, reference=str(path),
                       candidate=str(candidate_path),
                       native_rows=len(reference[0].rows),
                       candidate_rows=len(candidate[0].rows),
                       native_columns=len(reference[0].header),
                       candidate_columns=len(candidate[0].header),
                       native_sha256=hashlib.sha256(path.read_bytes()).hexdigest(),
                       candidate_sha256=hashlib.sha256(candidate_path.read_bytes()).hexdigest(),
                       differences=[str(value) for value in differences]))
receipt = dict(source_revision='95a98b0083aba2384c271e224746824fc9e40ce5',
               scope='Existing strict CellProfiler exported-table semantic comparison only; no image or ROI equivalence claim.',
               numeric_abs_tolerance=1e-6, numeric_rel_tolerance=1e-6,
               checks=checks)
(runs / 'perf-runtime-owner-fresh-native-table-parity-20261001.json').write_text(json.dumps(receipt, indent=2) + '\n')
print(json.dumps(receipt, indent=2))
assert not any(check['differences'] for check in checks)
